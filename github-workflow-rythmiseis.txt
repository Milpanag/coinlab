# Ενημερώνει καθημερινά την ενότητα «Από το ΕΚΠΑ» με τα νέα του hub.uoa.gr
name: Νέα ΕΚΠΑ
on:
  schedule:
    - cron: "0 5 * * *"   # κάθε μέρα στις 08:00 ώρα Ελλάδας (περίπου)
  workflow_dispatch:       # επιτρέπει χειροκίνητη εκτέλεση από την καρτέλα Actions
permissions:
  contents: write
jobs:
  fetch:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: python scripts/fetch_uoa_news.py
      - name: Αποθήκευση αλλαγών
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data/uoa-news.json
          git diff --cached --quiet || git commit -m "Ενημέρωση νέων ΕΚΠΑ"
          git push
